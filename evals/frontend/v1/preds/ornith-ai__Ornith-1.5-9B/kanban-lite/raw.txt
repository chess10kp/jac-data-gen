<script>
  function createCard(text) {
    const card = document.createElement('div');
    card.className = 'card';
    card.dataset.col = 'todo';
    card.textContent = text;
    const moveLeft = document.createElement('button');
    moveLeft.className = 'move-left';
    moveLeft.textContent = '\u2190';
    const moveRight = document.createElement('button');
    moveRight.className = 'move-right';
    moveRight.textContent = '\u2192';
    card.appendChild(moveLeft);
    card.appendChild(moveRight);
    updateButtons(card);
    return card;
  }

  function addCard(text) {
    const todoCol = document.querySelector('.column[data-col="todo"]');
    const card = createCard(text);
    todoCol.querySelector('.cards').appendChild(card);
    updateBadges();
  }

  function updateButtons(card) {
    const col = card.dataset.col;
    const moveLeft = card.querySelector('.move-left');
    const moveRight = card.querySelector('.move-right');
    if (col === 'todo') {
      moveLeft.disabled = true;
      moveRight.disabled = false;
    } else if (col === 'done') {
      moveLeft.disabled = false;
      moveRight.disabled = true;
    } else {
      moveLeft.disabled = false;
      moveRight.disabled = false;
    }
  }

  function moveCard(card, direction) {
    const currentCol = card.dataset.col;
    if (direction === 'right') {
      if (currentCol === 'done') return;
      card.dataset.col = 'done';
    } else {
      if (currentCol === 'todo') return;
      card.dataset.col = 'todo';
    }
    const col = document.querySelector('.column[data-col="' + card.dataset.col + '"]');
    col.querySelector('.cards').appendChild(card);
    updateButtons(card);
    updateBadges();
  }

  function updateBadges() {
    document.querySelectorAll('.column').forEach(function(col) {
      const count = col.querySelectorAll('.card').length;
      const badge = col.querySelector('.badge');
      if (badge) badge.textContent = count;
    });
  }

  document.getElementById('add-btn').addEventListener('click', function() {
    const input = document.getElementById('task-input');
    const text = input.value.trim();
    if (text) {
      addCard(text);
      input.value = '';
      input.focus();
    }
  });

  document.getElementById('task-input').addEventListener('keydown', function(e) {
    if (e.key === 'Enter') {
      document.getElementById('add-btn').click();
    }
  });

  document.querySelectorAll('.move-left').forEach(function(btn) {
    btn.addEventListener('click', function() {
      const card = this.closest('.card');
      moveCard(card, 'left');
    });
  });

  document.querySelectorAll('.move-right').forEach(function(btn) {
    btn.addEventListener('click', function() {
      const card = this.closest('.card');
      moveCard(card, 'right');
    });
  });

  updateBadges();
</script>
