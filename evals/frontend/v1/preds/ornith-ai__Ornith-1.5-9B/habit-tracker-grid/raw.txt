<script>
  (function () {
    var STORAGE_KEY = "habits";
    var SEED = [
      { name: "Water", color: "#60a5fa" },
      { name: "Exercise", color: "#f472b6" },
      { name: "Read", color: "#34d399" },
      { name: "Meditate", color: "#a78bfa" },
      { name: "Sleep", color: "#fbbf24" }
    ];

    function freshHabits() {
      return SEED.map(function (h) {
        return { name: h.name, color: h.color, days: [false, false, false, false, false, false, false] };
      });
    }

    function getHabits() {
      var raw = localStorage.getItem(STORAGE_KEY);
      if (raw) {
        try {
          var parsed = JSON.parse(raw);
          if (parsed && Array.isArray(parsed) && parsed.length) {
            return parsed;
          }
        } catch (e) {}
      }
      return freshHabits();
    }

    function saveHabits(habits) {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(habits));
    }

    function calculateStreak(days) {
      var streak = 0;
      for (var i = 0; i < days.length; i++) {
        if (days[i]) {
          streak++;
        } else {
          break;
        }
      }
      return streak;
    }

    function updateStreak(tr) {
      var inputs = tr.querySelectorAll("input[type=checkbox]");
      var streakSpan = tr.querySelector(".streak");
      var days = [];
      for (var i = 0; i < inputs.length; i++) {
        days.push(inputs[i].checked);
      }
      var streak = calculateStreak(days);
      streakSpan.textContent = streak;
      streakSpan.className = "streak" + (streak > 0 ? " active" : "");
    }

    function render() {
      var tbody = document.querySelector("#habit-table tbody");
      tbody.innerHTML = "";
      var habits = getHabits();
      habits.forEach(function (habit) {
        var tr = document.createElement("tr");

        var nameCell = document.createElement("td");
        nameCell.className = "habit-name";
        var dot = document.createElement("span");
        dot.className = "habit-dot";
        dot.style.background = habit.color || "#9ca3af";
        var label = document.createElement("span");
        label.textContent = habit.name;
        nameCell.appendChild(dot);
        nameCell.appendChild(label);
        tr.appendChild(nameCell);

        for (var d = 0; d < 7; d++) {
          var td = document.createElement("td");
          var cb = document.createElement("input");
          cb.type = "checkbox";
          cb.checked = !!habit.days[d];
          cb.addEventListener("change", function () {
            habit.days[d] = cb.checked;
            saveHabits(habits);
            updateStreak(tr);
          });
          td.appendChild(cb);
          tr.appendChild(td);
        }

        var streakCell = document.createElement("td");
        var streakSpan = document.createElement("span");
        streakSpan.className = "streak";
        updateStreak(tr);
        streakCell.appendChild(streakSpan);
        tr.appendChild(streakCell);

        tbody.appendChild(tr);
      });
    }

    document.getElementById("reset-btn").addEventListener("click", function () {
      saveHabits(freshHabits());
      render();
    });

    render();
  })();
</script>
