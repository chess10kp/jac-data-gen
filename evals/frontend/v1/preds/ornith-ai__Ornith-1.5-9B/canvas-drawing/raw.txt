  <h1>Drawing App</h1>
  <div class="toolbar">

      <span class="group-label">Color</span>

    </div>
    <div class="group size-group">
      <span class="group-label">Size</span>
      <input type="range" id="size" min="1" max="40" value="4">
      <span class="size-value" id="sizeValue">4</span>
    </div>
    <div class="group">
      <button id="undo">Undo</button>
      <button id="clear" class="clear">Clear</button>
    </div>
  </div>
  <div class="canvas-wrap">
    <canvas id="board" width="900" height="500"></canvas>
  </div>
  <script>
    const canvas = document.getElementById('board');
    const ctx = canvas.getContext('2d');
    const swatchesEl = document.getElementById('swatches');
    const sizeEl = document.getElementById('size');
    const sizeValueEl = document.getElementById('sizeValue');
    const undoBtn = document.getElementById('undo');
    const clearBtn = document.getElementById('clear');

    const colors = ['#000000', '#ef4444', '#f59e0b', '#22c55e', '#3b82f6', '#8b5cf6', '#ec4899'];
    let currentColor = colors[0];
    let currentSize = 4;
    let strokes = [];
    let currentStroke = null;
    let drawing = false;

    function render() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      for (const s of strokes) {
        if (s.points.length === 0) continue;
        ctx.strokeStyle = s.color;
        ctx.lineWidth = s.size;
        ctx.lineCap = 'round';
        ctx.lineJoin = 'round';
        ctx.beginPath();
        ctx.moveTo(s.points[0].x, s.points[0].y);
        for (let i = 1; i < s.points.length; i++) {
          ctx.lineTo(s.points[i].x, s.points[i].y);
        }
        ctx.stroke();
      }
    }

    function getPos(e) {
      const rect = canvas.getBoundingClientRect();
      const scaleX = canvas.width / rect.width;
      const scaleY = canvas.height / rect.height;
      const point = e.touches ? e.touches[0] : e;
      return {
        x: (point.clientX - rect.left) * scaleX,
        y: (point.clientY - rect.top) * scaleY
      };
    }

    function startStroke(e) {
      drawing = true;
      currentStroke = { points: [getPos(e)], color: currentColor, size: currentSize };
      render();
    }

    function moveStroke(e) {
      if (!drawing) return;
      const pos = getPos(e);
      currentStroke.points.push(pos);
      render();
      e.preventDefault();
    }

    function endStroke() {
      if (!drawing) return;
      drawing = false;
      if (currentStroke && currentStroke.points.length > 0) {
        strokes.push(currentStroke);
      }
      currentStroke = null;
      updateUndo();
    }

    function updateUndo() {
      undoBtn.disabled = strokes.length === 0;
    }

    function undo() {
      if (strokes.length === 0) return;
      strokes.pop();
      render();
      updateUndo();
    }

    function clearAll() {
      strokes = [];
      render();
      updateUndo();
    }

    colors.forEach((c, i) => {
      const el = document.createElement('div');
      el.className = 'swatch' + (i === 0 ? ' active' : '');
      el.style.background = c;
      el.addEventListener('click', () => {
        currentColor = c;
        document.querySelectorAll('.swatch').forEach(s => s.classList.remove('active'));
        el.classList.add('active');
      });
      swatchesEl.appendChild(el);
    });

    sizeEl.addEventListener('input', () => {
      currentSize = parseInt(sizeEl.value, 10);
      sizeValueEl.textContent = currentSize;
    });

    undoBtn.addEventListener('click', undo);
    clearBtn.addEventListener('click', clearAll);

    canvas.addEventListener('mousedown', startStroke);
    canvas.addEventListener('mousemove', moveStroke);
    window.addEventListener('mouseup', endStroke);
    canvas.addEventListener('touchstart', startStroke, { passive: false });
    canvas.addEventListener('touchmove', moveStroke, { passive: false });
    window.addEventListener('touchend', endStroke);

    render();
  </script>
