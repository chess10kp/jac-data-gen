<script>
  (function () {
    var STORAGE_KEY = "expenses";
    var CATEGORIES = ["Food", "Transport", "Fun", "Other"];
    var COLORS = { Food: "#3b82f6", Transport: "#10b981", Fun: "#f59e0b", Other: "#8b5cf6" };

    var form = document.getElementById("form");
    var amountInput = document.getElementById("amount");
    var categoryInput = document.getElementById("category");
    var noteInput = document.getElementById("note");
    var filterInput = document.getElementById("filter");
    var tbody = document.getElementById("tbody");
    var totalEl = document.getElementById("total");
    var chart = document.getElementById("chart");
    var legend = document.getElementById("legend");

    function load() {
      try {
        var raw = localStorage.getItem(STORAGE_KEY);
        return raw ? JSON.parse(raw) : [];
      } catch (e) { return []; }
    }
    function save(rows) {
      try { localStorage.setItem(STORAGE_KEY, JSON.stringify(rows)); } catch (e) {}
    }

    function fmt(n) {
      return "$" + n.toFixed(2);
    }

    function escapeHtml(s) {
      return String(s).replace(/[&<>"']/g, function (m) {
        return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[m];
      });
    }

    function sumFor(category, list) {
      return list.filter(function (r) { return r.category === category; }).reduce(function (a, r) { return a + r.amount; }, 0);
    }

    function render() {
      var rows = load();
      var filter = filterInput.value;
      var visible = rows.filter(function (r) { return filter === "All" || r.category === filter; });

      tbody.innerHTML = "";
      if (visible.length === 0) {
        var empty = document.createElement("tr");
        var td = document.createElement("td");
        td.className = "empty";
        td.colSpan = 4;
        td.textContent = "No expenses to show.";
        empty.appendChild(td);
        tbody.appendChild(empty);
      } else {
        visible.forEach(function (r) {
          var tr = document.createElement("tr");
          tr.innerHTML =
            '<td>' + escapeHtml(r.category) + '</td>' +
            '<td>' + escapeHtml(r.note || "") + '</td>' +
            '<td class="amount">' + fmt(r.amount) + '</td>' +
            '<td class="actions"><button class="del" data-idx="' + r.idx + '">Delete</button></td>';
          tbody.appendChild(tr);
        });
      }

      var total = 0;
      visible.forEach(function (r) { total += r.amount; });
      totalEl.textContent = fmt(total);

      renderChart(visible);
    }

    function renderChart(visible) {
      chart.innerHTML = "";
      var max = 0;
      CATEGORIES.forEach(function (c) {
        var t = sumFor(c, visible);
        if (t > max) max = t;
      });

      var W = 400, H = 200;
      var padTop = 20, padBottom = 30, padLeft = 10, padRight = 10;
      var chartW = W - padLeft - padRight;
      var chartH = H - padTop - padBottom;
      var barW = chartW / CATEGORIES.length;

      var baseline = document.createElementNS("http://www.w3.org/2000/svg", "line");
      baseline.setAttribute("x1", padLeft);
      baseline.setAttribute("y1", padTop + chartH);
      baseline.setAttribute("x2", padLeft + chartW);
      baseline.setAttribute("y2", padTop + chartH);
      baseline.setAttribute("stroke", "#e5e7eb");
      baseline.setAttribute("stroke-width", "1");
      chart.appendChild(baseline);

      CATEGORIES.forEach(function (c, i) {
        var t = sumFor(c, visible);
        var h = max > 0 ? (t / max) * chartH : 0;
        var x = padLeft + i * barW + 2;
        var y = padTop + chartH - h;

        var rect = document.createElementNS("http://www.w3.org/2000/svg", "rect");
        rect.setAttribute("x", x);
        rect.setAttribute("y", y);
        rect.setAttribute("width", barW - 4);
        rect.setAttribute("height", Math.max(h, 0));
        rect.setAttribute("rx", 3);
        rect.setAttribute("fill", COLORS[c]);
        chart.appendChild(rect);

        var val = document.createElementNS("http://www.w3.org/2000/svg", "text");
        val.setAttribute("x", x + (barW - 4) / 2);
        val.setAttribute("y", Math.max(y - 4, padTop + 14));
        val.setAttribute("text-anchor", "middle");
        val.setAttribute("font-size", "11");
        val.setAttribute("fill", "#52606d");
        val.textContent = fmt(t);
        chart.appendChild(val);

        var label = document.createElementNS("http://www.w3.org/2000/svg", "text");
        label.setAttribute("x", x + (barW - 4) / 2);
        label.setAttribute("y", H - 10);
        label.setAttribute("text-anchor", "middle");
        label.setAttribute("font-size", "11");
        label.setAttribute("fill", "#52606d");
        label.textContent = c;
        chart.appendChild(label);
      });

      legend.innerHTML = "";
      CATEGORIES.forEach(function (c) {
        var span = document.createElement("span");
        span.innerHTML = '<span class="swatch" style="background:' + COLORS[c] + '"></span>' + c;
        legend.appendChild(span);
      });
    }

    form.addEventListener("submit", function (e) {
      e.preventDefault();
      var amount = parseFloat(amountInput.value);
      if (isNaN(amount) || amount < 0) return;
      var rows = load();
      rows.push({
        idx: Date.now() + "-" + Math.random().toString(36).slice(2, 6),
        category: categoryInput.value,
        amount: amount,
        note: noteInput.value.trim()
      });
      save(rows);
      amountInput.value = "";
      noteInput.value = "";
      render();
    });

    tbody.addEventListener("click", function (e) {
      var btn = e.target.closest("button.del");
      if (!btn) return;
      var idx = btn.getAttribute("data-idx");
      var rows = load();
      var newRows = rows.filter(function (r) { return r.idx !== idx; });
      save(newRows);
      render();
    });

    filterInput.addEventListener("change", render);

    render();
  })();
</script>
