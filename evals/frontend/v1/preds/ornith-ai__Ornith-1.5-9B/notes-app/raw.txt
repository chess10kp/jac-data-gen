<script>
  (function () {
    var STORAGE_KEY = "notes";
    var notes = [];
    var activeNoteId = null;
    var activeTag = "";
    var searchQuery = "";

    var searchInput = document.getElementById("search");
    var tagFilter = document.getElementById("tagFilter");
    var noteList = document.getElementById("noteList");
    var newNoteBtn = document.getElementById("newNote");
    var deleteNoteBtn = document.getElementById("deleteNote");
    var editorEmpty = document.getElementById("editorEmpty");
    var editorContent = document.getElementById("editorContent");
    var titleInput = document.getElementById("titleInput");
    var tagsInput = document.getElementById("tagsInput");
    var bodyInput = document.getElementById("bodyInput");
    var saveHint = document.getElementById("saveHint");

    function genId() {
      return Date.now().toString(36) + Math.random().toString(36).slice(2);
    }

    function loadNotes() {
      var data = localStorage.getItem(STORAGE_KEY);
      if (data) {
        try { notes = JSON.parse(data); } catch (e) { notes = []; }
      }
      if (!notes.length) {
        notes.push({
          id: genId(),
          title: "Welcome Note",
          body: "Start writing here. Your notes auto-save as you type.",
          tags: [],
          createdAt: Date.now(),
          updatedAt: Date.now()
        });
      }
    }

    function saveNotes() {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(notes));
    }

    function parseTags(str) {
      return str.split(",")
        .map(function (t) { return t.trim(); })
        .filter(function (t) { return t.length > 0; });
    }

    function getFilteredNotes() {
      var result = notes;
      if (activeTag) {
        result = result.filter(function (n) { return n.tags.indexOf(activeTag) !== -1; });
      }
      if (searchQuery) {
        var q = searchQuery.toLowerCase();
        result = result.filter(function (n) {
          return (n.title || "").toLowerCase().indexOf(q) !== -1 ||
                 (n.body || "").toLowerCase().indexOf(q) !== -1;
        });
      }
      return result;
    }

    function getAllTags() {
      var set = {};
      notes.forEach(function (n) {
        n.tags.forEach(function (t) { set[t] = true; });
      });
      return Object.keys(set).sort();
    }

    function renderTagFilter() {
      var tags = getAllTags();
      tagFilter.innerHTML = '<option value="">All tags</option>';
      tags.forEach(function (t) {
        var opt = document.createElement("option");
        opt.value = t;
        opt.textContent = t;
        tagFilter.appendChild(opt);
      });
    }

    function renderNoteList() {
      var filtered = getFilteredNotes();
      noteList.innerHTML = "";
      if (filtered.length === 0) {
        var empty = document.createElement("div");
        empty.style.padding = "24px";
        empty.style.textAlign = "center";
        empty.style.color = "#9aa0b5";
        empty.style.fontSize = "13px";
        empty.textContent = "No notes found";
        noteList.appendChild(empty);
      } else {
        filtered.forEach(function (n) {
          var item = document.createElement("div");
          item.className = "note-item" + (n.id === activeNoteId ? " active" : "");
          item.onclick = (function (id) { return function () { openNote(id); }; })(n.id);

          var title = n.title || "(untitled)";
          var titleEl = document.createElement("div");
          titleEl.className = "note-item-title";
          titleEl.textContent = title;

          var preview = n.body ? n.body.slice(0, 60) : "";
          var previewEl = document.createElement("div");
          previewEl.className = "note-item-preview";
          previewEl.textContent = preview;

          if (n.tags.length) {
            var tagsEl = document.createElement("div");
            tagsEl.className = "note-item-tags";
            n.tags.slice(0, 4).forEach(function (t) {
              var pill = document.createElement("span");
              pill.className = "tag-pill";
              pill.textContent = t;
              tagsEl.appendChild(pill);
            });
            item.appendChild(tagsEl);
          }

          item.appendChild(titleEl);
          item.appendChild(previewEl);
          noteList.appendChild(item);
        });
      }
      deleteNoteBtn.disabled = (activeNoteId === null);
    }

    function openNote(id) {
      activeNoteId = id;
      var note = notes.find(function (n) { return n.id === id; });
      if (!note) return;
      titleInput.value = note.title;
      tagsInput.value = note.tags.join(", ");
      bodyInput.value = note.body;
      editorEmpty.style.display = "none";
      editorContent.style.display = "flex";
      deleteNoteBtn.disabled = false;
      renderNoteList();
    }

    function createNote() {
      var note = {
        id: genId(),
        title: "",
        body: "",
        tags: [],
        createdAt: Date.now(),
        updatedAt: Date.now()
      };
      notes.unshift(note);
      saveNotes();
      openNote(note.id);
      titleInput.focus();
    }

    function deleteNote() {
      if (activeNoteId === null) return;
      notes = notes.filter(function (n) { return n.id !== activeNoteId; });
      activeNoteId = null;
      saveNotes();
      editorEmpty.style.display = "";
      editorContent.style.display = "none";
      deleteNoteBtn.disabled = true;
      renderNoteList();
    }

    function saveActiveNote() {
      if (activeNoteId === null) return;
      var note = notes.find(function (n) { return n.id === activeNoteId; });
      if (!note) return;
      note.title = titleInput.value;
      note.body = bodyInput.value;
      note.tags = parseTags(tagsInput.value);
      note.updatedAt = Date.now();
      saveNotes();
      renderNoteList();
    }

    function onInput() {
      saveActiveNote();
    }

    searchInput.addEventListener("input", function () {
      searchQuery = searchInput.value;
      renderNoteList();
    });
    tagFilter.addEventListener("change", function () {
      activeTag = tagFilter.value;
      renderNoteList();
    });
    newNoteBtn.addEventListener("click", createNote);
    deleteNoteBtn.addEventListener("click", deleteNote);
    titleInput.addEventListener("input", onInput);
    tagsInput.addEventListener("input", onInput);
    bodyInput.addEventListener("input", onInput);

    loadNotes();
    renderTagFilter();
    renderNoteList();
  })();
</script>
