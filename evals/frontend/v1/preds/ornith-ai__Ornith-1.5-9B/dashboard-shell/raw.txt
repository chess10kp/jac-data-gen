<script>
  (function () {
    var navLinks = document.querySelectorAll('.sidebar nav a');
    var pages = document.querySelectorAll('.page');
    var pageTitle = document.getElementById('page-title');
    var welcomeText = document.getElementById('welcome-text');

    var pageTitles = {
      overview: 'Overview',
      data: 'Data',
      settings: 'Settings'
    };

    var pageWelcome = {
      overview: 'Welcome back 👋',
      data: 'Your data overview',
      settings: 'Manage your preferences'
    };

    function switchPage(pageId) {
      navLinks.forEach(function (link) {
        link.classList.toggle('active', link.getAttribute('data-page') === pageId);
      });
      pages.forEach(function (page) {
        page.classList.toggle('active', page.id === 'page-' + pageId);
      });
      pageTitle.textContent = pageTitles[pageId] || pageId;
      welcomeText.textContent = pageWelcome[pageId] || '';
      window.scrollTo(0, 0);
    }

    navLinks.forEach(function (link) {
      link.addEventListener('click', function (e) {
        e.preventDefault();
        switchPage(link.getAttribute('data-page'));
      });
    });

    // Settings form
    var form = document.getElementById('settings-form');
    var displayName = document.getElementById('display-name');
    var themeSelect = document.getElementById('theme-select');
    var savedMsg = document.getElementById('saved-msg');

    function loadSettings() {
      var saved = JSON.parse(localStorage.getItem('dashboard-settings') || '{}');
      if (saved.displayName) displayName.value = saved.displayName;
      if (saved.theme) themeSelect.value = saved.theme;
    }

    function saveSettings() {
      var settings = {
        displayName: displayName.value.trim(),
        theme: themeSelect.value
      };
      localStorage.setItem('dashboard-settings', JSON.stringify(settings));
      savedMsg.classList.add('show');
      setTimeout(function () {
        savedMsg.classList.remove('show');
      }, 2500);
    }

    form.addEventListener('submit', function (e) {
      e.preventDefault();
      saveSettings();
    });

    loadSettings();
  })();
</script>
