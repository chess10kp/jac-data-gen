  <script>
    (function() {
      var form = document.getElementById('wizardForm');
      var submitted = document.getElementById('submitted');
      var steps = document.querySelectorAll('.step');
      var progressSteps = document.querySelectorAll('.progress-step');
      var progressLines = document.querySelectorAll('.progress-line');
      var currentStep = 1;

      var nameInput = document.getElementById('name');
      var emailInput = document.getElementById('email');
      var nameError = document.getElementById('nameError');
      var emailError = document.getElementById('emailError');
      var planError = document.getElementById('planError');

      var radioOptions = document.querySelectorAll('.radio-option');

      function getSelectedPlan() {
        var checked = document.querySelector('input[name="plan"]:checked');
        return checked ? checked.value : null;
      }

      function updateRadioSelection() {
        radioOptions.forEach(function(opt) {
          var input = opt.querySelector('input');
          if (input.checked) opt.classList.add('selected');
          else opt.classList.remove('selected');
        });
      }

      radioOptions.forEach(function(opt) {
        opt.addEventListener('click', function(e) {
          if (e.target.tagName === 'INPUT') {
            e.preventDefault();
            e.target.checked = true;
            updateRadioSelection();
          }
        });
      });

      function validateStep1() {
        var valid = true;

        var name = nameInput.value.trim();
        if (name === '') {
          nameInput.classList.add('invalid');
          nameError.classList.add('show');
          valid = false;
        } else {
          nameInput.classList.remove('invalid');
          nameError.classList.remove('show');
        }

        var email = emailInput.value.trim();
        if (email === '' || email.indexOf('@') === -1) {
          emailInput.classList.add('invalid');
          emailError.classList.add('show');
          valid = false;
        } else {
          emailInput.classList.remove('invalid');
          emailError.classList.remove('show');
        }

        return valid;
      }

      function validateStep2() {
        var plan = getSelectedPlan();
        if (!plan) {
          planError.classList.add('show');
          return false;
        }
        planError.classList.remove('show');
        return true;
      }

      function showStep(step) {
        currentStep = step;
        steps.forEach(function(s) {
          s.classList.toggle('active', parseInt(s.getAttribute('data-step')) === step);
        });

        progressSteps.forEach(function(ps) {
          var psStep = parseInt(ps.getAttribute('data-step'));
          ps.classList.toggle('active', psStep === step);
          ps.classList.toggle('done', psStep < step);
        });
        progressLines.forEach(function(line, i) {
          line.classList.toggle('done', i < step - 1);
        });
      }

      function next() {
        if (currentStep === 1) {
          if (validateStep1()) showStep(2);
        } else if (currentStep === 2) {
          if (validateStep2()) {
            showStep(3);
            updateSummary();
          }
        }
      }

      function back() {
        if (currentStep === 2) showStep(1);
        else if (currentStep === 3) showStep(2);
      }

      function updateSummary() {
        document.getElementById('summaryName').textContent = nameInput.value.trim();
        document.getElementById('summaryEmail').textContent = emailInput.value.trim();
        document.getElementById('summaryPlan').textContent = getSelectedPlan();
      }

      document.getElementById('nextBtn').addEventListener('click', next);
      document.getElementById('nextBtn2').addEventListener('click', next);
      document.getElementById('backBtn').addEventListener('click', back);
      document.getElementById('backBtn2').addEventListener('click', back);
      document.getElementById('submitBtn').addEventListener('click', function() {
        submitted.classList.add('show');
        form.style.display = 'none';
      });
    })();
  </script>
